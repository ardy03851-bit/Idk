/* ---------------- persistence ---------------- */
const persist = debounce(() => {
  try {
    localStorage.setItem(KEY, JSON.stringify({ files: FILES, entry: entryPath, tabs: openTabs, active: activePath }));
  } catch (e) {
    toast('Storage full — project not saved', 'err');
  }
}, 700);

function loadProject(){
  let raw = null;
  try { raw = localStorage.getItem(KEY); } catch (e) {}
  if (raw) {
    try {
      const d = JSON.parse(raw);
      if (d && d.files && Object.keys(d.files).length) {
        FILES = d.files;
        entryPath = d.entry || pickEntry();
        openTabs = (d.tabs || []).filter(t => FILES[t]);
        activePath = (d.active && FILES[d.active]) ? d.active : (openTabs[0] || null);
        return;
      }
    } catch (e) {}
  }
  FILES = JSON.parse(JSON.stringify(STARTER));
  entryPath = 'index.html';
  openTabs = ['index.html'];
  activePath = 'index.html';
}

function pickEntry(){
  const htmls = Object.keys(FILES).filter(p => /\.html?$/i.test(p));
  if (htmls.includes('index.html')) return 'index.html';
  return htmls[0] || Object.keys(FILES)[0] || null;
}

function fileIconInfo(path){
  const e = extOf(path);
  if (['png','jpg','jpeg','gif','webp','svg','avif','bmp','ico'].includes(e)) return ['image','c-img'];
  if (['mp4','webm','mov','ogv'].includes(e)) return ['film','c-media'];
  if (['mp3','wav','ogg','m4a'].includes(e)) return ['music','c-media'];
  if (['zip','rar','7z','tar','gz'].includes(e)) return ['archive','c-zip'];
  if (['html','htm'].includes(e)) return ['code','c-html'];
  if (['css','scss','less'].includes(e)) return ['css','c-css'];
  if (['js','mjs','ts','jsx','tsx','json'].includes(e)) return ['code','c-js'];
  return ['file','c-file'];
}

function buildTree(paths){
  const root = { children:{}, files:[] };
  for (const p of paths){
    const parts = p.split('/');
    let node = root;
    for (let i = 0; i < parts.length - 1; i++){
      const name = parts[i];
      if (!node.children[name]) node.children[name] = { children:{}, files:[], path: parts.slice(0, i+1).join('/') };
      node = node.children[name];
    }
    node.files.push({ name: parts[parts.length-1], path: p });
  }
  return root;
}

function renderTree(){
  const tree = $('#tree');
  const scroll = tree.scrollTop;
  const paths = Object.keys(FILES).sort();
  const root = buildTree(paths);
  let html = '';

  function walkDir(node, depth){
    const dirs = Object.keys(node.children).sort((a,b)=>a.localeCompare(b));
    for (const d of dirs){
      const child = node.children[d];
      const isCollapsed = collapsedDirs.has(child.path);
      html += '
' +
        '' + icon('chev', 12) + '' +
        '' + icon('folder', 14) + '' +
        '' + esc(d) + '' +
        '' +
          '' +
          '' +
        '
';
      if (!isCollapsed) walkDir(child, depth + 1);
    }
    node.files.sort((a,b)=>a.name.localeCompare(b.name));
    for (const f of node.files){
      const [ic, cls] = fileIconInfo(f.path);
      const active = f.path === activePath ? ' active' : '';
      html += '
' +
        '' + icon(ic, 14) + '' +
        '' + esc(f.name) + '' +
        (f.path === entryPath ? 'ENTRY' : '') +
        '' +
        '
';
    }
  }
  walkDir(root, 0);
  tree.innerHTML = html || '
No files yet
';
  tree.scrollTop = scroll;
}

$('#tree').addEventListener('click', e => {
  const dirEl = e.target.closest('[data-dir]');
  const fileEl = e.target.closest('[data-file]');
  const actBtn = e.target.closest('[data-act]');

  if (actBtn){
    e.stopPropagation();
    const act = actBtn.dataset.act, p = actBtn.dataset.path;
    if (act === 'filedel'){ deletePath(p); }
    if (act === 'deldir'){ deleteDir(p); }
    if (act === 'newfile-here'){ newFile(p); }
    return;
  }
  if (fileEl){ openFile(fileEl.dataset.file); return; }
  if (dirEl){
    const p = dirEl.dataset.dir;
    if (collapsedDirs.has(p)) collapsedDirs.delete(p); else collapsedDirs.add(p);
    renderTree();
  }
});

function uniquePath(p){
  if (!FILES[p]) return p;
  const dot = p.lastIndexOf('.');
  const base = dot < 0 ? p : p.slice(0, dot);
  const ext = dot < 0 ? '' : p.slice(dot);
  let i = 2;
  while (FILES[base + '-' + i + ext]) i++;
  return base + '-' + i + ext;
}

function newFile(parentDir){
  const name = prompt('New file name:', 'untitled.html');
  if (!name) return;
  const p = uniquePath((parentDir ? parentDir + '/' : '') + name.trim().replace(/^\/+/, ''));
  FILES[p] = { c: '' };
  renderTree();
  openFile(p);
  persist();
}

function deletePath(p){
  if (!confirm('Delete ' + p + '?')) return;
  delete FILES[p];
  openTabs = openTabs.filter(t => t !== p);
  if (activePath === p) activePath = openTabs[0] || null;
  if (entryPath === p) entryPath = pickEntry();
  renderTree(); renderTabs(); loadIntoEditor(); schedulePreview(); persist();
}

function deleteDir(dir){
  const victims = Object.keys(FILES).filter(p => p.startsWith(dir + '/'));
  if (!victims.length) return;
  if (!confirm('Delete folder "' + dir + '" and ' + victims.length + ' file(s)?')) return;
  victims.forEach(p => delete FILES[p]);
  openTabs = openTabs.filter(t => !victims.includes(t));
  if (!FILES[activePath]) activePath = openTabs[0] || null;
  if (!FILES[entryPath]) entryPath = pickEntry();
  renderTree(); renderTabs(); loadIntoEditor(); schedulePreview(); persist();
}

function renamePath(p){
  const n = prompt('Rename to:', p);
  if (!n || n === p) return;
  const np = n.trim().replace(/^\/+/, '');
  if (FILES[np]) { toast('That name already exists', 'err'); return; }
  FILES[np] = FILES[p];
  delete FILES[p];
  openTabs = openTabs.map(t => t === p ? np : t);
  if (activePath === p) activePath = np;
  if (entryPath === p) entryPath = np;
  renderTree(); renderTabs(); loadIntoEditor(); schedulePreview(); persist();
}

function duplicatePath(p){
  const np = uniquePath(p.replace(/(\.[^./]+)$/, '-copy$1'));
  FILES[np] = { c: FILES[p].c, b: FILES[p].b };
  renderTree(); persist();
  toast('Duplicated', 'ok');
}

function downloadBlob(blob, filename){
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = filename;
  document.body.appendChild(a);
  a.click();
  setTimeout(() => { URL.revokeObjectURL(url); a.remove(); }, 1500);
}

function downloadFile(p){
  const f = FILES[p];
  if (!f) return;
  if (f.b){
    const comma = f.c.indexOf(',');
    const bytes = b64decodeToBytes(f.c.slice(comma + 1));
    downloadBlob(new Blob([bytes], { type: mimeFor(p) }), baseName(p));
  } else {
    downloadBlob(new Blob([f.c], { type: mimeFor(p) }), baseName(p));
  }
}

function openFile(p){
  if (!FILES[p]) return;
  if (!openTabs.includes(p)) openTabs.push(p);
  activePath = p;
  renderTabs(); renderTree(); loadIntoEditor();
  persist();
}

function closeTab(p){
  openTabs = openTabs.filter(t => t !== p);
  if (activePath === p) activePath = openTabs[openTabs.length - 1] || null;
  renderTabs(); renderTree(); loadIntoEditor();
  persist();
}

function renderTabs(){
  const bar = $('#tabbar');
  bar.innerHTML = openTabs.map(p => {
    const [ic, cls] = fileIconInfo(p);
    return '
' +
      '' + icon(ic, 13) + '' +
      '' + esc(baseName(p)) + '' +
      '' + icon('x', 12) + '
';
  }).join('');
}

$('#tabbar').addEventListener('click', e => {
  const closeBtn = e.target.closest('[data-close]');
  if (closeBtn){ closeTab(closeBtn.dataset.close); return; }
  const tab = e.target.closest('[data-tab]');
  if (tab) openFile(tab.dataset.tab);
});

function loadIntoEditor(){
  if (!activePath || !FILES[activePath]){
    codeEl.value = '';
    hlEl.innerHTML = '';
    gutterInner.textContent = '1';
    binaryView.style.display = 'none';
    codeEl.style.visibility = 'hidden';
    return;
  }
  const f = FILES[activePath];
  if (f.b || !isText(activePath)){
    binaryView.style.display = 'flex';
    codeEl.style.visibility = 'hidden';
    const mime = mimeFor(activePath);
    let inner = '';
    if (mime.startsWith('image/')) inner = '';
    else if (mime.startsWith('video/')) inner = '';
    else if (mime.startsWith('audio/')) inner = '';
    else inner = '
' + icon('file', 40) + '
';
    binaryView.innerHTML = inner + '
' + esc(activePath) + '
' + mime + '
';
    return;
  }
  binaryView.style.display = 'none';
  codeEl.style.visibility = 'visible';
  codeEl.value = f.c;
  renderHighlight();
  codeEl.scrollTop = 0; codeEl.scrollLeft = 0;
  hlEl.scrollTop = 0; hlEl.scrollLeft = 0;
  gutterInner.style.transform = 'translateY(0px)';
}

const jsRe = new RegExp(

  '|(\\b\\d[\\w.]*\\b)' +                                                     // 3 number

  '|([A-Za-z_$][\\w$]*)(?=\\s*\\()' +                                         // 5 function

function hlJS(src){
  let out = '', last = 0, m;
  jsRe.lastIndex = 0;
  while ((m = jsRe.exec(src))){
    if (m.index > last) out += esc(src.slice(last, m.index));
    const t = m[0];
    if (m[1]) out += '' + esc(t) + '';
    else if (m[2]) out += '' + esc(t) + '';
    else if (m[3]) out += '' + esc(t) + '';
    else if (m[4]) out += '' + esc(t) + '';
    else if (m[5]) out += '' + esc(t) + '';
    else out += '' + esc(t) + '';
    last = m.index + t.length;
  }
  out += esc(src.slice(last));
  return out;
}
